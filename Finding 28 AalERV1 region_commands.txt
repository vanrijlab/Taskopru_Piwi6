#These lines of commands show how 28 AalERV1 regions were found. 
#You need to GCF_006496715.1_Aalbo_primary.1_genomic.fna in the directory. 

#First make a blast directory
makeblastdb -in GCF_006496715.1_Aalbo_primary.1_genomic.fna -dbtype nucl -out GCF_006496715.1_Aalbo_primary.1_genomic.db
#Retrive the FASTA sequence of the originally identified AalERV1 locus (NW_021837045.1_146399653-146407556) from the genome.
bedtools getfasta -fi GCF_006496715.1_Aalbo_primary.1_genomic.fna -bed NW_021837045.1:146399653-146407556.bed > NW_021837045.1:146399653-146407556.fa
#BLASTn search against the same genome, output is named three here, as there are expected to be three ORFs. 
blastn -query NW_021837045.1:146399653-146407556.fa -db GCF_006496715.1_Aalbo_primary.1_genomic.db -out three.out -outfmt 6
#sort according to the length of the results
cat three.out | sort -k4n > sorted_length_three.out
#filter the ones with a minimum length of 7000 nucleotides
cat sorted_length_three.out | awk '$4>7000 {print$0}' |awk 'BEGIN{OFS="\t"}{print $2, $9, $10}' > three.bed
#if the elements are in antisense orientation, where the start site ($2) is bigger than the end sites, reverse the coordinates.  
awk '$2 > $3 { temp = $3; $3 = $2; $2 = temp } 1' OFS='\t' three.bed > orderedthree.bed
